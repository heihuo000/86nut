## 🧠 NUT脚本系统示例

### 示例5: 基础技能脚本

```
// 火球术技能定义
ENUM_FIREBALL_SUBSTATE_START <- 0;
ENUM_FIREBALL_SUBSTATE_CAST  <- 1;
ENUM_FIREBALL_SUBSTATE_END   <- 2;

// 技能检查函数
function checkExecutableSkill_Fireball(obj)
{
    if(!obj) return false;
    
    // 检查是否正在使用火球术技能
    local b_useskill = obj.sq_IsUseSkill(SKILL_FIREBALL);
    if(b_useskill) {
        obj.sq_IntVectClear();  // 清空状态数组
        obj.sq_IntVectPush(ENUM_FIREBALL_SUBSTATE_START);  // 设置初始状态
        obj.sq_AddSetStatePacket(STATE_FIREBALL, STATE_PRIORITY_USER, true);  // 设置状态
        return true;
    }
    return false;
}

// 状态设置函数
function onSetState_Fireball(obj, state, datas, isResetTimer)
{
    if(!obj) return;
    
    local level = sq_GetSkillLevel(obj, SKILL_FIREBALL);  // 获取技能等级
    local castTime = sq_GetIntData(obj, SKILL_FIREBALL, 0, level);  // 施法时间
    local damage = sq_GetIntData(obj, SKILL_FIREBALL, 1, level);    // 伤害值
    
    // 根据不同状态执行不同逻辑
    switch(state) {
        case ENUM_FIREBALL_SUBSTATE_START:
            // 准备阶段：播放准备动画
            obj.sq_PlayAnimation("fireball_prepare");
            break;
        case ENUM_FIREBALL_SUBSTATE_CAST:
            // 施法阶段：创建火球特效并造成伤害
            obj.sq_CreateEffect("fireball_effect", 100, 0, 50);
            obj.sq_ApplyDamage(damage);
            break;
        case ENUM_FIREBALL_SUBSTATE_END:
            // 结束阶段：清理状态
            obj.sq_ClearState(STATE_FIREBALL);
            break;
    }
}

// 状态更新函数
function onUpdate_Fireball(obj, deltaTime)
{
    if(!obj) return;
    
    local currentState = obj.sq_IntVectGet(0);  // 获取当前状态
    
    // 根据当前状态更新逻辑
    switch(currentState) {
        case ENUM_FIREBALL_SUBSTATE_START:
            // 准备时间结束后进入施法阶段
            if(obj.sq_GetStateTime() > 1000) {  // 1秒准备时间
                obj.sq_IntVectSet(0, ENUM_FIREBALL_SUBSTATE_CAST);
                onSetState_Fireball(obj, ENUM_FIREBALL_SUBSTATE_CAST, null, false);
            }
            break;
        case ENUM_FIREBALL_SUBSTATE_CAST:
            // 施法时间结束后进入结束阶段
            if(obj.sq_GetStateTime() > 500) {  // 0.5秒施法时间
                obj.sq_IntVectSet(0, ENUM_FIREBALL_SUBSTATE_END);
                onSetState_Fireball(obj, ENUM_FIREBALL_SUBSTATE_END, null, false);
            }
            break;
    }
}
```

#### 功能拆解
- 三阶段技能执行（准备→施法→结束）
- 状态机管理技能流程
- 根据技能等级调整参数
- 动画、特效、伤害一体化实现

### 示例6: 复合技能脚本

```
// 冰霜新星技能定义
ENUM_FROST_NOVA_SUBSTATE_START <- 0;
ENUM_FROST_NOVA_SUBSTATE_CAST  <- 1;
ENUM_FROST_NOVA_SUBSTATE_END   <- 2;

// 冰霜状态定义
STATE_FROST_DEBUFF <- 1001;

// 技能检查函数
function checkExecutableSkill_FrostNova(obj)
{
    if(!obj) return false;
    
    local b_useskill = obj.sq_IsUseSkill(SKILL_FROST_NOVA);
    if(b_useskill) {
        obj.sq_IntVectClear();
        obj.sq_IntVectPush(ENUM_FROST_NOVA_SUBSTATE_START);
        obj.sq_AddSetStatePacket(STATE_FROST_NOVA, STATE_PRIORITY_USER, true);
        
        // 同时施加自身加速效果
        obj.sq_AddBuff(STATE_HASTE, 3000, 1.5);  // 3秒1.5倍加速
        
        return true;
    }
    return false;
}

// 状态设置函数
function onSetState_FrostNova(obj, state, datas, isResetTimer)
{
    if(!obj) return;
    
    local level = sq_GetSkillLevel(obj, SKILL_FROST_NOVA);
    local radius = sq_GetIntData(obj, SKILL_FROST_NOVA, 0, level);    // 范围
    local damage = sq_GetIntData(obj, SKILL_FROST_NOVA, 1, level);    // 伤害
    local freezeDuration = sq_GetIntData(obj, SKILL_FROST_NOVA, 2, level); // 冰冻时间
    
    switch(state) {
        case ENUM_FROST_NOVA_SUBSTATE_START:
            obj.sq_PlayAnimation("frostnova_prepare");
            obj.sq_CreateEffect("frostnova_aura", 0, 0, 0);
            break;
            
        case ENUM_FROST_NOVA_SUBSTATE_CAST:
            obj.sq_PlayAnimation("frostnova_cast");
            obj.sq_CreateEffect("frostnova_explosion", 0, 0, 0);
            
            // 对周围敌人造成伤害和冰冻效果
            local targets = obj.sq_GetTargetsInRange(radius);
            foreach(target in targets) {
                target.sq_ApplyDamage(damage);
                target.sq_AddDebuff(STATE_FROST_DEBUFF, freezeDuration, 0.5);  // 减速50%
            }
            break;
            
        case ENUM_FROST_NOVA_SUBSTATE_END:
            obj.sq_ClearState(STATE_FROST_NOVA);
            break;
    }
}

// 状态更新函数
function onUpdate_FrostNova(obj, deltaTime)
{
    if(!obj) return;
    
    local currentState = obj.sq_IntVectGet(0);
    
    switch(currentState) {
        case ENUM_FROST_NOVA_SUBSTATE_START:
            if(obj.sq_GetStateTime() > 800) {
                obj.sq_IntVectSet(0, ENUM_FROST_NOVA_SUBSTATE_CAST);
                onSetState_FrostNova(obj, ENUM_FROST_NOVA_SUBSTATE_CAST, null, false);
            }
            break;
            
        case ENUM_FROST_NOVA_SUBSTATE_CAST:
            if(obj.sq_GetStateTime() > 600) {
                obj.sq_IntVectSet(0, ENUM_FROST_NOVA_SUBSTATE_END);
                onSetState_FrostNova(obj, ENUM_FROST_NOVA_SUBSTATE_END, null, false);
            }
            break;
    }
}
```

    switch(state) {
        case ENUM_FLAME_STORM_SUBSTATE_PREPARE:
            obj.sq_PlayAnimation("FlameStorm_Prepare");
            obj.sq_CreateEffect("flame_aura", 0, 0, 30);
            break;
            
        case ENUM_FLAME_STORM_SUBSTATE_CAST:
            obj.sq_PlayAnimation("FlameStorm_Cast");
            obj.sq_CreateEffect("flame_storm", 0, 0, 0);
            
            // 造成范围伤害
            local targets = obj.sq_GetTargetsInRange(radius);
            foreach(target in targets) {
                target.sq_ApplyDamageOverTime(damage, duration, 500);  // 每0.5秒造成伤害
                target.sq_AddDebuff(STATE_BURN, duration, 1.0);      // 燃烧效果
            }
            
            // 击退效果
            obj.sq_KnockbackTargets(radius, 200);  // 击退力度200
            break;
            
        case ENUM_FLAME_STORM_SUBSTATE_END:
            obj.sq_ClearState(STATE_FLAME_STORM);
            obj.sq_RemoveBuff(STATE_INVINCIBLE);
            break;
    }
}

function onUpdate_FlameStorm(obj, deltaTime)
{
    if(!obj) return;
    
    local currentState = obj.sq_IntVectGet(0);
    
    switch(currentState) {
        case ENUM_FLAME_STORM_SUBSTATE_PREPARE:
            if(obj.sq_GetStateTime() > 1500) {  // 1.5秒准备时间
                obj.sq_IntVectSet(0, ENUM_FLAME_STORM_SUBSTATE_CAST);
                onSetState_FlameStorm(obj, ENUM_FLAME_STORM_SUBSTATE_CAST, null, false);
            }
            break;
            
        case ENUM_FLAME_STORM_SUBSTATE_CAST:
            if(obj.sq_GetStateTime() > 3000) {  // 3秒施法时间
                obj.sq_IntVectSet(0, ENUM_FLAME_STORM_SUBSTATE_END);
                onSetState_FlameStorm(obj, ENUM_FLAME_STORM_SUBSTATE_END, null, false);
            }
            break;
    }
}

function onUpdate_UltimateBoss(obj, deltaTime)
{
    if(!obj) return;
    
    local healthPercent = (obj.sq_GetCurrentHealth().tofloat() / obj.sq_GetMaxHealth().tofloat()) * 100;
    local currentPhase = obj.sq_GetPhase();
    
    // 根据血量切换阶段
    if(healthPercent < 30 && currentPhase != ENUM_BOSS_PHASE_FINAL) {
        obj.sq_SetPhase(ENUM_BOSS_PHASE_FINAL);
        enterFinalPhase(obj);
    }
    else if(healthPercent < 70 && currentPhase == ENUM_BOSS_PHASE_NORMAL) {
        obj.sq_SetPhase(ENUM_BOSS_PHASE_RAGE);
        enterRageMode(obj);
    }
    
    // 执行阶段特定行为
    switch(currentPhase) {
        case ENUM_BOSS_PHASE_NORMAL:
            executeNormalPhaseActions(obj, deltaTime);
            break;
        case ENUM_BOSS_PHASE_RAGE:
            executeRagePhaseActions(obj, deltaTime);
            break;
        case ENUM_BOSS_PHASE_FINAL:
            executeFinalPhaseActions(obj, deltaTime);
            break;
    }
}
